<script>

  let sceneClicked = false;


  function showStage(stageId) {

    document.querySelectorAll('.stage').forEach(s => s.classList.remove('active'));

    document.getElementById(stageId).classList.add('active');

  }


  function handleSceneClick() {

    if (!sceneClicked) {

      sceneClicked = true;

      const dog = document.getElementById('dog');

      const speechBubble = document.getElementById('speechBubble');

      const sceneHint = document.getElementById('sceneHint');

      

      // Запускаем анимацию собаки

      dog.classList.add('jump');

      

      // Скрываем подсказку

      sceneHint.style.display = 'none';

      

      // Через 2 секунды показываем облачко с текстом

      setTimeout(() => {

        speechBubble.classList.add('show');

      }, 2000);

      

      // Через 4 секунды переходим к выбору эмоций

      setTimeout(() => {

        showStage('stage-emotion');

      }, 4000);

    }

  }


  const correctEmotion = 'fear';

  const emotionHints = {

    anger: 'Подумай ещё! Это не злость. Какая эмоция помогает быть осторожным?',

    sadness: 'Не совсем! Грусть — это про другое. Подумай про опасность!',

    fear: '',

    joy: 'Радость — это когда всё хорошо! А тут что-то неожиданное...'

  };


  document.querySelectorAll('#stage-emotion .emotion-btn').forEach(btn => {

    btn.addEventListener('click', function() {

      const emotion = this.dataset.emotion;

      const feedback = document.getElementById('feedback-emotion');

      if (emotion === correctEmotion) {

        this.classList.add('correct');

        feedback.className = 'feedback success';

        feedback.textContent = '✅ Верно! Это Страх!';

        setTimeout(() => showStage('stage-power'), 1500);

      } else {

        this.classList.add('wrong');

        feedback.className = 'feedback hint';

        feedback.textContent = '💡 ' + emotionHints[emotion];

        setTimeout(() => this.classList.remove('wrong'), 500);

      }

    });

  });


  const correctPower = 'siren';

  const powerHints = {

    siren: '',

    teddy: 'Мишка 🧸 — это уютно, но он не спасёт от опасности. Подумай про сигнал тревоги!',

    burger: 'Бургер 🍔 вкусный, но он не суперсила страха. Попробуй ещё!'

  };


  document.querySelectorAll('#stage-power .power-btn').forEach(btn => {

    btn.addEventListener('click', function() {

      const power = this.dataset.power;

      const feedback = document.getElementById('feedback-power');

      if (power === correctPower) {

        this.classList.add('correct');

        feedback.className = 'feedback success';

        feedback.textContent = '🚨 Супер! Сирена — суперсила Страха!';

        setTimeout(() => showStage('stage-final'), 1500);

      } else {

        this.classList.add('wrong');

        feedback.className = 'feedback hint';

        feedback.textContent = '💡 ' + powerHints[power];

        setTimeout(() => this.classList.remove('wrong'), 500);

      }

    });

  });

</script>
